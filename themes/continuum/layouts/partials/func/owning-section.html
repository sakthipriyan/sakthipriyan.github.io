{{- /* Returns the top-level section page that owns the given page, resolving taxonomy and term pages back to the section that declares that taxonomy in its front matter. Returns false when no section owns the page (e.g. the home page). Usage: {{ $section := partial "func/owning-section.html" . }} */ -}} {{- $page := . -}} {{- $result := false -}} {{- range $page.Site.Sections -}} {{- $s := . -}} {{- if not $result -}} {{- if or (eq $page.Section $s.Section) (and $s.Params.taxonomy (eq $page.Section $s.Params.taxonomy)) -}} {{- $result = $s -}} {{- end -}} {{- end -}} {{- end -}} {{- return $result -}}
